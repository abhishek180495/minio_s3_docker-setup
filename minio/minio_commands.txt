1. Pull the latest stable image of AIStor Server
    docker pull quay.io/minio/aistor/minio
2. Run the AIStor Server in a container

    docker run -dt ^
  -p 9000:9000 -p 9001:9001 ^
  -v C:/Users/Priya Madhur/OneDrive/Desktop/abhi_project/minio-docker-container/minio/data:/mnt/data ^
  -v C:/Users/Priya Madhur/OneDrive/Desktop/abhi_project/minio-docker-container/minio/license/minio.license:/minio.license ^
  -v C:/Users/Priya Madhur/OneDrive/Desktop/abhi_project/minio-docker-container/minio/certs:/etc/minio/certs ^
  --name aistor-server ^
  quay.io/minio/minio:latest server /mnt/data --console-address ":9001" --license /minio.license
